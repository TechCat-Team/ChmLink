cmake_minimum_required(VERSION 3.16)
project(chmlink LANGUAGES C)

add_library(chmlink_packet STATIC src/packet/packet.c)
target_include_directories(chmlink_packet PUBLIC include)
target_compile_features(chmlink_packet PUBLIC c_std_99)

add_library(chmlink_protocol STATIC src/protocol/batch.c)
target_link_libraries(chmlink_protocol PUBLIC chmlink_packet)
target_compile_features(chmlink_protocol PUBLIC c_std_99)

include(CTest)
if(BUILD_TESTING)
    add_executable(packet_test tests/packet_test.c)
    target_link_libraries(packet_test PRIVATE chmlink_packet)
    add_test(NAME packet_test COMMAND packet_test)
    add_executable(batch_test tests/batch_test.c)
    target_link_libraries(batch_test PRIVATE chmlink_protocol)
    add_test(NAME batch_test COMMAND batch_test)
endif()
