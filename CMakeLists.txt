cmake_minimum_required(VERSION 3.16)
project(chmlink LANGUAGES C)

add_library(chmlink_packet STATIC src/packet/packet.c)
target_include_directories(chmlink_packet PUBLIC include)
target_compile_features(chmlink_packet PUBLIC c_std_99)

add_library(chmlink_protocol STATIC src/protocol/batch.c src/protocol/query.c src/protocol/session.c src/packet/offload.c src/packet/split.c src/packet/gro.c)

# Shared-memory packet ring, used by the SDK and (as source) by the driver.
add_library(chmlink_ring STATIC src/ring/ring.c)
target_include_directories(chmlink_ring PUBLIC include)
target_compile_features(chmlink_ring PUBLIC c_std_99)

# The protocol library uses both the packet helpers and the ring layout.
target_link_libraries(chmlink_protocol PUBLIC chmlink_packet chmlink_ring)
target_compile_features(chmlink_protocol PUBLIC c_std_99)

if(WIN32)
    # The SDK: chmlink.dll (exports in src/windows/chmlink.def) for
    # applications, and the same code as a static library for chmlink-device,
    # which must run on its own.
    set(CHMLINK_SDK_SOURCES src/windows/adapter.c src/windows/receive.c src/windows/send.c
        src/windows/wait.c src/windows/manage.c src/windows/device_config.c src/windows/package.c src/windows/log.c src/windows/coalesce.c)
    set(CHMLINK_SDK_LIBRARIES iphlpapi ole32 setupapi newdev uuid advapi32 bcrypt)
    add_library(chmlink_static STATIC ${CHMLINK_SDK_SOURCES})
    target_link_libraries(chmlink_static PUBLIC chmlink_protocol ${CHMLINK_SDK_LIBRARIES})
    target_include_directories(chmlink_static PUBLIC include)
    target_compile_definitions(chmlink_static PRIVATE _WIN32_WINNT=0x0A00)
    add_library(chmlink SHARED ${CHMLINK_SDK_SOURCES} src/windows/chmlink.def)
    target_link_libraries(chmlink PRIVATE chmlink_protocol ${CHMLINK_SDK_LIBRARIES})
    target_include_directories(chmlink PUBLIC include)
    target_compile_definitions(chmlink PRIVATE _WIN32_WINNT=0x0A00)
    # Version resource from version.h, the single source of the release.
    enable_language(RC)
    file(READ include/chmlink/version.h CHMLINK_VERSION_HEADER)
    string(REGEX MATCH "CHMLINK_VERSION_STRING \"([0-9.]+)\"" _ "${CHMLINK_VERSION_HEADER}")
    set(CHMLINK_VERSION "${CMAKE_MATCH_1}")
    string(REPLACE "." "," CHMLINK_VERSION_COMMAS "${CHMLINK_VERSION}")
    configure_file(src/windows/version.rc.in version.rc @ONLY)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS include/chmlink/version.h)
    target_sources(chmlink PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/version.rc)
    # A directory with the signed chmlink.inf, .sys and .cat embeds that
    # package, so applications ship chmlink.dll alone.
    set(CHMLINK_DRIVER_DIR "" CACHE PATH "Signed driver package to embed in chmlink.dll")
    if(CHMLINK_DRIVER_DIR)
        file(TO_CMAKE_PATH "${CHMLINK_DRIVER_DIR}" CHMLINK_DRIVER_DIR)
        set(CHMLINK_DRIVER_FILES)
        foreach(name chmlink.inf chmlink.sys chmlink.cat)
            if(NOT EXISTS "${CHMLINK_DRIVER_DIR}/${name}")
                message(FATAL_ERROR "CHMLINK_DRIVER_DIR has no ${name}")
            endif()
            list(APPEND CHMLINK_DRIVER_FILES "${CHMLINK_DRIVER_DIR}/${name}")
        endforeach()
        configure_file(src/windows/driver.rc.in driver.rc @ONLY)
        set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/driver.rc PROPERTIES
            OBJECT_DEPENDS "${CHMLINK_DRIVER_FILES}")
        target_sources(chmlink PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/driver.rc)
    endif()
    # cmake --install: what an application needs, chmlink.dll (bin), its
    # import library (lib) and the public headers (include/chmlink).
    install(TARGETS chmlink RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
    install(FILES include/chmlink/api.h include/chmlink/manage.h include/chmlink/version.h
                  include/chmlink/packet.h include/chmlink/protocol.h include/chmlink/offload.h
                  include/chmlink/gro.h
            DESTINATION include/chmlink)
    add_executable(chmlink-list examples/list.c)
    target_link_libraries(chmlink-list PRIVATE chmlink)
    add_executable(chmlink-capture examples/capture.c)
    target_link_libraries(chmlink-capture PRIVATE chmlink)
    add_executable(chmlink-inject examples/inject.c)
    target_link_libraries(chmlink-inject PRIVATE chmlink)
    target_compile_definitions(chmlink-inject PRIVATE _CRT_SECURE_NO_WARNINGS)
    add_executable(chmlink-quickstart examples/quickstart.c)
    target_link_libraries(chmlink-quickstart PRIVATE chmlink iphlpapi ws2_32)
    # Driver-package, concurrency and surprise-removal tests, through chmlink.dll
    # with the driver embedded as applications ship it (tools/test-package.ps1).
    add_executable(chmlink-package-test tests/windows/package_test.c)
    target_link_libraries(chmlink-package-test PRIVATE chmlink setupapi)
    if(MINGW)
        target_link_options(chmlink-package-test PRIVATE -municode)
    endif()
    add_executable(chmlink-device tools/device.c)
    target_include_directories(chmlink-device PRIVATE src/windows)
    target_compile_definitions(chmlink-device PRIVATE _WIN32_WINNT=0x0A00)
    target_link_libraries(chmlink-device PRIVATE chmlink_static)
    if(MINGW)
        target_link_options(chmlink-device PRIVATE -municode)
    endif()
    add_executable(chmlink-arp-test tests/windows/arp_test.c tests/support/arp_peer.c)
    target_include_directories(chmlink-arp-test PRIVATE tests/support)
    target_link_libraries(chmlink-arp-test PRIVATE chmlink iphlpapi ws2_32)
    add_executable(chmlink-ip-test tests/windows/ip_test.c)
    target_link_libraries(chmlink-ip-test PRIVATE chmlink iphlpapi ws2_32)
    add_executable(chmlink-bench bench/packet_bench.c)
    target_link_libraries(chmlink-bench PRIVATE chmlink iphlpapi ws2_32)
    add_executable(chmlink-tcp-bench bench/tcp_bench.c tests/support/arp_peer.c)
    target_include_directories(chmlink-tcp-bench PRIVATE tests/support)
    target_link_libraries(chmlink-tcp-bench PRIVATE chmlink iphlpapi ws2_32)
    # The same benchmarks over Wintun (bench/wintun_shim.c), for comparison;
    # they load wintun.dll from their own directory at run time.
    add_executable(wintun-bench bench/packet_bench.c bench/wintun_shim.c)
    target_link_libraries(wintun-bench PRIVATE chmlink_protocol iphlpapi ws2_32)
    add_executable(wintun-tcp-bench bench/tcp_bench.c bench/wintun_shim.c tests/support/arp_peer.c)
    target_include_directories(wintun-tcp-bench PRIVATE tests/support)
    target_link_libraries(wintun-tcp-bench PRIVATE chmlink_protocol iphlpapi ws2_32)
    # The same benchmarks over tap-windows6 (bench/tap_shim.c) in TUN mode;
    # they need one installed TAP-Windows Adapter V9.
    add_executable(tap-bench bench/packet_bench.c bench/tap_shim.c)
    target_link_libraries(tap-bench PRIVATE chmlink_protocol iphlpapi ws2_32 ole32)
    add_executable(tap-tcp-bench bench/tcp_bench.c bench/tap_shim.c tests/support/arp_peer.c)
    target_include_directories(tap-tcp-bench PRIVATE tests/support)
    target_link_libraries(tap-tcp-bench PRIVATE chmlink_protocol iphlpapi ws2_32 ole32)
    add_executable(chmlink-lifecycle-test tests/windows/lifecycle_test.c tests/windows/queue_test.c tests/windows/sdk_test.c)
    target_link_libraries(chmlink-lifecycle-test PRIVATE chmlink iphlpapi ws2_32)
    if(MINGW)
        target_link_options(chmlink-lifecycle-test PRIVATE -municode)
    endif()
    add_executable(chmlink-power-test tests/windows/power_test.c)
    target_link_libraries(chmlink-power-test PRIVATE chmlink powrprof)
    if(MINGW)
        target_link_options(chmlink-power-test PRIVATE -municode)
    endif()
    add_executable(chmlink-ioctl-fuzz tests/windows/ioctl_fuzz.c)
    target_link_libraries(chmlink-ioctl-fuzz PRIVATE chmlink_protocol chmlink_ring)
    if(MINGW)
        target_link_options(chmlink-ioctl-fuzz PRIVATE -municode)
    endif()
endif()
