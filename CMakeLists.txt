cmake_minimum_required(VERSION 3.16)
project(chmlink LANGUAGES C)

add_library(chmlink_packet STATIC src/packet/packet.c)
target_include_directories(chmlink_packet PUBLIC include)
target_compile_features(chmlink_packet PUBLIC c_std_99)

add_library(chmlink_protocol STATIC src/protocol/batch.c)
target_link_libraries(chmlink_protocol PUBLIC chmlink_packet)
target_compile_features(chmlink_protocol PUBLIC c_std_99)

if(WIN32)
    add_library(chmlink STATIC src/windows/api.c)
    target_link_libraries(chmlink PUBLIC chmlink_protocol iphlpapi ole32)
    target_compile_definitions(chmlink PRIVATE _WIN32_WINNT=0x0A00)
    add_executable(chmlink-list examples/list.c)
    target_link_libraries(chmlink-list PRIVATE chmlink)
    add_executable(chmlink-capture examples/capture.c)
    target_link_libraries(chmlink-capture PRIVATE chmlink)
    add_executable(chmlink-inject examples/inject.c)
    target_link_libraries(chmlink-inject PRIVATE chmlink)
    target_compile_definitions(chmlink-inject PRIVATE _CRT_SECURE_NO_WARNINGS)
    add_executable(chmlink-device tools/device.c tools/device_config.c)
    target_include_directories(chmlink-device PRIVATE include)
    target_compile_definitions(chmlink-device PRIVATE _WIN32_WINNT=0x0A00)
    target_link_libraries(chmlink-device PRIVATE setupapi newdev uuid advapi32 iphlpapi ole32 bcrypt)
    if(MINGW)
        target_link_options(chmlink-device PRIVATE -municode)
    endif()
    add_executable(chmlink-arp-test tests/windows_arp_test.c tests/arp_peer.c)
    target_link_libraries(chmlink-arp-test PRIVATE chmlink iphlpapi ws2_32)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(config_test tests/config_test.c)
    target_include_directories(config_test PRIVATE include)
    add_test(NAME config_test COMMAND config_test)
    add_executable(packet_test tests/packet_test.c)
    target_link_libraries(packet_test PRIVATE chmlink_packet)
    add_test(NAME packet_test COMMAND packet_test)
    add_executable(batch_test tests/batch_test.c)
    target_link_libraries(batch_test PRIVATE chmlink_protocol)
    add_test(NAME batch_test COMMAND batch_test)
    add_executable(arp_peer_test tests/arp_peer_test.c tests/arp_peer.c)
    add_test(NAME arp_peer_test COMMAND arp_peer_test)
endif()
